// IMPORTANT DESIGN RULE
// Do not show another brand's chemical bottle as if it were a Proline product.

// Proline/local media always has priority.
// -----------------------------------------------------------------------------

function normalize(value: string) {
  return value
    .normalize("NFD")
    .replace(/[\u0300-\u036f]/g, "")
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, " ")
    .replace(/\s+/g, " ")
    .trim();
}

function classify(name: string) {
  const hay = normalize(name);
  for (const rule of CATEGORY_RULES) {
    if (rule.terms.some((term) => hay.includes(normalize(term)))) return rule.category;
  }

  // These short product codes do not carry enough semantic information.
  // Treat them as general professional products rather than inventing a use.
  if (["3s", "dinol", "dinot pro", "dinot pro 1"].some((x) => hay === x || hay.startsWith(`${x} `))) {
    return "Professionnel";
  }
  return "Professionnel";
}

function visualKind(name: string, category: string) {
  const hay = normalize(name);

  if (hay.includes("lave vaisselle") || hay.includes("lave verre") || hay.includes("lave-verre")) return "machine";
  if (hay.includes("machine")) return "machine";
  if (hay.includes("dinol") || hay.includes("dinot") || hay === "3s") return "communs";
  if (hay.includes("degraissant plaque") || hay.includes("degraissant cuisine") || hay.includes("rincage plaque")) return "cuisine";
  if (category === "Linge") return "linge";
  if (category === "Piscine") return "piscine";
  if (category === "Hygiène") return "sanitaire";
  if (hay.includes("lave vitre") || hay.includes("lave-vitre")) return "vitre";
  if (hay.includes("marbre") || hay.includes("shine") || hay.includes("sticker") || hay.includes("degrissol")) return "communs";
  if (category === "Parfums") return "hotel";
  if (category === "Entretien") return "communs";
  return "hotel";
}

async function walk(dir: string): Promise<string[]> {
  const result: string[] = [];
  try {
    const entries = await readdir(dir, { withFileTypes: true });
    for (const entry of entries) {
      const absolute = path.join(dir, entry.name);
      if (entry.isDirectory()) result.push(...await walk(absolute));
      else if (IMAGE_EXTENSIONS.has(path.extname(entry.name).toLowerCase())) result.push(absolute);
    }
  } catch {
    // Missing media folder should not prevent product loading.
  }
  return result;
}

function toPublicPath(absolutePath: string) {
  const relative = path
    .relative(path.join(process.cwd(), "public"), absolutePath)
    .split(path.sep)
    .join("/");
  return `/${encodeURI(relative).replace(/%2F/gi, "/")}`;
}

function stableIndex(value: string) {
  return normalize(value)
    .split("")
    .reduce((total, char) => (total * 31 + char.charCodeAt(0)) >>> 0, 7);
}

function chooseLocal(
  kind: keyof typeof LOCAL_VISUALS,
  imageEntries: string[],
  productName: string,
) {
  const candidates = LOCAL_VISUALS[kind];
  const normalizedCandidates = new Map(
    imageEntries.map((absolute) => [normalize(path.basename(absolute)), absolute]),
  );

  // Exact curated file first.
  const exact = candidates
    .map((name) => normalizedCandidates.get(normalize(name)))
    .find(Boolean);
  if (exact) return exact;

  // Then allow recursive media matches by the visual-family filename keywords.
  const familyTerms: Record<string, string[]> = {
    cuisine: ["cuisine"],
    machine: ["machine", "lave vaisselle"],
    linge: ["buanderie", "bauderie", "linge"],
    sanitaire: ["sanitaire"],
    communs: ["locaux commun", "commun"],
    piscine: ["piscine"],
    hotel: ["chambre", "quarto", "room", "hotel", "nettoyage"],
  };
  const terms = familyTerms[kind] ?? [];
  const ranked = imageEntries.filter((absolute) => {
    const hay = normalize(absolute);
    return terms.some((term) => hay.includes(normalize(term)));
  });

  if (!ranked.length) return null;
  return ranked[stableIndex(productName) % ranked.length];
}

function webFallback(kind: keyof typeof WEB_VISUALS | "hotel" | "communs") {
  if (kind in WEB_VISUALS) return WEB_VISUALS[kind as keyof typeof WEB_VISUALS];
  return WEB_VISUALS.cuisine;
}

export async function GET() {
  try {
    const [pdfEntries, imageEntries] = await Promise.all([
      readdir(PRODUCT_DIR, { withFileTypes: true }),
      walk(MEDIA_DIR),
    ]);

    const images = imageEntries.sort((a, b) => a.localeCompare(b));

    const products = pdfEntries
      .filter((entry) => entry.isFile() && entry.name.toLowerCase().endsWith(".pdf"))
      .map((entry) => {
        const name = entry.name.replace(/\.pdf$/i, "");
        const category = classify(name);
        const kind = visualKind(name, category) as keyof typeof LOCAL_VISUALS;
        const local = chooseLocal(kind, images, name);

        const hasWeb = (key: string): key is keyof typeof WEB_VISUALS => key in WEB_VISUALS;
        const webKey =
          kind === "vitre" ? "vitre" :
          kind === "piscine" ? "piscine" :
          kind === "linge" ? "linge" :
          kind === "machine" ? "machine" :
          kind === "sanitaire" ? "sanitaire" :
          kind === "cuisine" ? "cuisine" :
          "cuisine";

        const photo = local ? toPublicPath(local) : hasWeb(webKey) ? WEB_VISUALS[webKey] : webFallback("cuisine");
        const fallbackPhoto = hasWeb(webKey) ? WEB_VISUALS[webKey] : WEB_VISUALS.cuisine;

        return {
          id: normalize(name).replace(/\s+/g, "-") || crypto.randomUUID(),
          name,
          file: entry.name,
          url: `/Produits/${encodeURIComponent(entry.name)}`,
          category,
          visualKind: kind,
          photo,
          fallbackPhoto,
          photoAlt: `${name} — illustration d'application professionnelle ${category}`,
          photoSource: local ? "local-category" : "web-category",
        };
      })
      .sort((a, b) => a.name.localeCompare(b.name, "fr"));

    return NextResponse.json(
      {
        products,
        count: products.length,
        localImages: images.length,
        mediaRoot: "/media",
        rule: "local-first-contextual-visuals",
      },
      { headers: { "Cache-Control": "no-store" } },
    );
  } catch (error) {
    console.error("/api/products:", error);
    return NextResponse.json(
      { products: [], count: 0, localImages: 0, error: "Impossible de lire les produits." },
      { status: 500 },
    );
  }
}
